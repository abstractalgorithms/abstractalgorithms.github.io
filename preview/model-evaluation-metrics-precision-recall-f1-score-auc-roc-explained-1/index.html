Model Evaluation Metrics: Precision, Recall, F1-Score, AUC-ROC Explained

Scenario context

🎯 Accuracy is a lie when classes are imbalanced. Real ML evaluation uses precision (how many positives are actually positive), recall (how many actual positives we caught), F1 (their balance), and AUC-ROC (performance across all thresholds). The right metric depends on your cost function: optimize precision for spam filtering, recall for cancer screening.

# Understanding metric computation internals
import time

def manual_precision_recall(y_true, y_pred):
    """Manual implementation to show the core logic"""
    tp = np.sum((y_true == 1) & (y_pred == 1))
    fp = np.sum((y_true == 0) & (y_pred == 1))
    fn = np.sum((y_true == 1) & (y_pred == 0))

    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0

    return precision, recall

# Performance comparison
n_samples = 1000000
y_true_large = np.random.choice([0, 1], n_samples, p=[0.98, 0.02])
y_pred_large = np.random.choice([0, 1], n_samples, p=[0.97, 0.03])

# Time sklearn implementation
start = time.time()
sklearn_p = precision_score(y_true_large, y_pred_large)
sklearn_r = recall_score(y_true_large, y_pred_large)
sklearn_time = time.time() - start

# Time manual implementation
start = time.time()
manual_p, manual_r = manual_precision_recall(y_true_large, y_pred_large)
manual_time = time.time() - start

print(f"Large dataset ({n_samples:,} samples):")
print(f"sklearn: P={sklearn_p:.4f}, R={sklearn_r:.4f}, Time={sklearn_time:.4f}s")
print(f"Manual:  P={manual_p:.4f}, R={manual_r:.4f}, Time={manual_time:.4f}s")
print(f"Speedup: {manual_time/sklearn_time:.1f}x (sklearn is faster due to C optimizations)")

def analyze_metric_complexity(n_samples_range):
    """Analyze how metric computation scales with dataset size"""

    results = []

    for n in n_samples_range:
        # Generate synthetic data
        y_true = np.random.choice([0, 1], n, p=[0.9, 0.1])
        y_pred = np.random.choice([0, 1], n, p=[0.88, 0.12])
        y_proba = np.random.random(n)

        # Time different metrics
        metrics_to_test = {
            'precision': lambda: precision_score(y_true, y_pred),
            'recall': lambda: recall_score(y_true, y_pred), 
            'f1': lambda: f1_score(y_true, y_pred),
            'accuracy': lambda: (y_true == y_pred).mean(),
            'auc_roc': lambda: roc_auc_score(y_true, y_proba),
            'confusion_matrix': lambda: confusion_matrix(y_true, y_pred)
        }

        sample_result = {'n_samples': n}

        for metric_name, metric_func in metrics_to_test.items():
            start = time.time()
            metric_func()
            elapsed = time.time() - start
            sample_result[f'{metric_name}_time'] = elapsed

        results.append(sample_result)

    return pd.DataFrame(results)

# Test complexity scaling
sample_sizes = [1000, 10000, 100000, 500000]
complexity_df = analyze_metric_complexity(sample_sizes)

print("Metric Computation Time Complexity:")
print(complexity_df.round(5))

# Show Big-O complexity
print("\nTime Complexity Summary:")
print("Precision/Recall/F1: O(n) - linear scan")  
print("Accuracy: O(n) - elementwise comparison")
print("AUC-ROC: O(n log n) - sorting required for curve")
print("Confusion Matrix: O(n) - counting operations")

def comprehensive_evaluation(model, X, y, threshold=0.5):
    """Complete model evaluation with cross-validation and multiple metrics"""

    # Stratified cross-validation setup
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)

    # Cross-validate multiple metrics
    cv_results = {
        'accuracy': cross_val_score(model, X, y, cv=skf, scoring='accuracy'),
        'f1': cross_val_score(model, X, y, cv=skf, scoring='f1'),
        'precision': cross_val_score(model, X, y, cv=skf, scoring='precision'),
        'recall': cross_val_score(model, X, y, cv=skf, scoring='recall'),
        'roc_auc': cross_val_score(model, X, y, cv=skf, scoring='roc_auc')
    }

    # Print cross-validation results
    print("5-Fold Cross-Validation Results:")
    print("=" * 40)
    for metric, scores in cv_results.items():
        print(f"{metric.capitalize():10}: {scores.mean():.3f} ± {scores.std():.3f}")

    # Single train-test evaluation for detailed analysis
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.3, random_state=42, stratify=y
    )

    model.fit(X_train, y_train)
    y_pred = model.predict(X_test)
    y_pred_proba = model.predict_proba(X_test)[:, 1]

    # Detailed single-split results
    print(f"\nSingle Test Split Results:")
    print("=" * 40)
    print(classification_report(y_test, y_pred, digits=3))

    return cv_results, (X_test, y_test, y_pred, y_pred_proba)

def production_monitoring_metrics(y_true, y_pred, y_pred_proba):
    """Key metrics for production ML monitoring"""

    metrics = {
        'precision': precision_score(y_true, y_pred),
        'recall': recall_score(y_true, y_pred), 
        'f1': f1_score(y_true, y_pred),
        'auc_roc': roc_auc_score(y_true, y_pred_proba),
        'prediction_rate': y_pred.mean(),  # % of predictions that are positive
        'actual_rate': y_true.mean(),      # % of actuals that are positive
        'calibration_gap': abs(y_pred_proba.mean() - y_true.mean())
    }

    return metrics

Failure and degraded modes

Precision and Recall** capture the two sides of classification performance:

  • Detection: Precision = TP / (TP + FP) — "Of all cases I predicted positive, how many were actually positive?" Recall = TP / (TP + FN) — "Of all actual positive cases, how many did I catch?"
  • User or operator impact: Precision = TP / (TP + FP) — "Of all cases I predicted positive, how many were actually positive?" Recall = TP / (TP + FN) — "Of all actual positive cases, how many did I catch?"
