import io.github.resilience4j.circuitbreaker.annotation.CircuitBreaker;
import io.github.resilience4j.ratelimiter.annotation.RateLimiter;
import org.springframework.stereotype.Service;
import org.springframework.web.client.HttpClientErrorException;
import org.springframework.web.client.RestTemplate;

@Service
public class OrderProcessingProducerService {

    private final RestTemplate restTemplate;
    private static final String CONSUMER_SERVICE = "consumerService"; // Resilience4j instance name

    public OrderProcessingProducerService(RestTemplate restTemplate) {
        this.restTemplate = restTemplate;
    }

    @CircuitBreaker(name = CONSUMER_SERVICE, fallbackMethod = "processOrderFallback")
    @RateLimiter(name = CONSUMER_SERVICE, fallbackMethod = "processOrderFallback")
    public String processOrder(String orderDetails) {
        try {
            // Simulate calling a downstream consumer service
            String response = restTemplate.postForObject("http://consumer-service/api/orders", orderDetails, String.class);
            System.out.println("Order processed successfully by consumer: " + response);
            return "Order processed: " + response;
        } catch (HttpClientErrorException.TooManyRequests e) {
            // Explicitly handle 429s from downstream as a backpressure signal
            System.err.println("Consumer service signaled backpressure (429). Falling back or retrying later.");
            throw e; // Let CircuitBreaker/RateLimiter handle the state transition
        }
    }

    private String processOrderFallback(String orderDetails, Throwable t) {
        System.err.println("Fallback triggered for order: " + orderDetails + ". Reason: " + t.getMessage());
        // Implement alternative logic, e.g., queueing for later, notifying admin, or returning a default response
        return "Order processing temporarily unavailable. Please try again later.";
    }
}

import io.resilience4j.circuitbreaker.CircuitBreakerConfig;
import io.resilience4j.core.registry.EntryAddedEvent;
import io.resilience4j.core.registry.EntryRemovedEvent;
import io.resilience4j.core.registry.EntryReplacedEvent;
import io.resilience4j.core.registry.EventConsumer;
import io.resilience4j.micrometer.tagged.TaggedCircuitBreakerMetrics;
import io.resilience4j.micrometer.tagged.TaggedRateLimiterMetrics;
import io.resilience4j.micrometer.tagged.TaggedBulkheadMetrics;
import io.resilience4j.micrometer.tagged.TaggedRetryMetrics;
import io.resilience4j.micrometer.tagged.TaggedTimeLimiterMetrics;
import io.resilience4j.circuitbreaker.CircuitBreakerRegistry;
import io.resilience4j.ratelimiter.RateLimiterConfig;
import io.resilience4j.ratelimiter.RateLimiterRegistry;
import io.resilience4j.bulkhead.BulkheadConfig;
import io.resilience4j.bulkhead.BulkheadRegistry;
import io.micrometer.core.instrument.MeterRegistry;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.time.Duration;

@Configuration
public class ResilienceConfig {

    @Bean
    public CircuitBreakerRegistry circuitBreakerRegistry(MeterRegistry meterRegistry) {
        CircuitBreakerConfig config = CircuitBreakerConfig.custom()
                .failureRateThreshold(30) // Circuit opens if 30% of calls fail
                .waitDurationInOpenState(Duration.ofSeconds(15)) // Stay open for 15 seconds
                .slidingWindowType(CircuitBreakerConfig.SlidingWindowType.COUNT_BASED)
                .slidingWindowSize(100) // Evaluate last 100 calls
                .minimumNumberOfCalls(10) // Minimum calls before evaluation starts
                .build();

        CircuitBreakerRegistry registry = CircuitBreakerRegistry.of(config);

        // Optional: Register metrics for monitoring
        TaggedCircuitBreakerMetrics.ofCircuitBreakerRegistry(registry).bindTo(meterRegistry);

        return registry;
    }

    @Bean
    public RateLimiterRegistry rateLimiterRegistry(MeterRegistry meterRegistry) {
        RateLimiterConfig config = RateLimiterConfig.custom()
                .limitForPeriod(5) // Allow 5 requests per period
                .limitRefreshPeriod(Duration.ofSeconds(1)) // Refresh every 1 second
                .timeoutDuration(Duration.ofSeconds(0)) // Fail fast if no permit available
                .build();

        RateLimiterRegistry registry = RateLimiterRegistry.of(config);
        TaggedRateLimiterMetrics.ofRateLimiterRegistry(registry).bindTo(meterRegistry);
        return registry;
    }

    @Bean
    public BulkheadRegistry bulkheadRegistry(MeterRegistry meterRegistry) {
        BulkheadConfig config = BulkheadConfig.custom()
                .maxConcurrentCalls(10) // Allow 10 concurrent calls
                .maxWaitDuration(Duration.ofMillis(0)) // Fail fast if no permit available
                .build();

        BulkheadRegistry registry = BulkheadRegistry.of(config);
        TaggedBulkheadMetrics.ofBulkheadRegistry(registry).bindTo(meterRegistry);
        return registry;
    }
}
